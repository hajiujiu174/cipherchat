cmake_minimum_required(VERSION 3.21)
project(CipherChatDesktopLauncher LANGUAGES CXX)
set_source_files_properties(../../resources/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/app.ico")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_add_executable(CipherChat-UI-Selector WIN32 main.cpp ../../resources/app.rc)
target_link_libraries(CipherChat-UI-Selector PRIVATE Qt6::Widgets)
if(MSVC)
    target_compile_options(CipherChat-UI-Selector PRIVATE /utf-8)
endif()
