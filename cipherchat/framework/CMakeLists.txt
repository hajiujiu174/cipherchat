set(CMAKE_CXX_STANDARD 20)

set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CCFW_BUILD_GUI "Build the CipherChat Qt desktop client" OFF)
option(CCFW_BUILD_RHINE_NATIVE "Build the Qt Rhine frontend with full animation" ON)
option(CCFW_BUILD_GUI_TESTS "Build Qt interaction and layout tests" ${CCFW_BUILD_GUI})
option(CCFW_BUILD_DELIVERY_TESTS "Register tests for existing delivery artifacts" ${CCFW_BUILD_GUI})
if(CCFW_BUILD_GUI_TESTS AND NOT CCFW_BUILD_GUI)
    message(FATAL_ERROR "CCFW_BUILD_GUI_TESTS requires CCFW_BUILD_GUI=ON")
endif()
option(CCFW_LINK_EXISTING "Compile and link existing algorithms for implementation tasks" ON)
# SessionService 已使用原算法，所有构建都必须链接唯一的实现。
if(NOT CCFW_LINK_EXISTING)
    set(CCFW_LINK_EXISTING ON CACHE BOOL "Required by implemented services" FORCE)
endif()
find_package(SQLite3 REQUIRED)
if(NOT TARGET SQLite3::SQLite3)
    add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
get_filename_component(CIPHERCHAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

add_library(ccfw_contracts INTERFACE)
target_include_directories(ccfw_contracts INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CIPHERCHAT_ROOT}"
    "${CIPHERCHAT_ROOT}/third_party")
if(MSVC)
    target_compile_options(ccfw_contracts INTERFACE /utf-8 /W4)
else()
    target_compile_options(ccfw_contracts INTERFACE -Wall -Wextra -Wpedantic)
endif()

# 原算法源文件只有一份。后续填充涉及算法调用时开启该选项，禁止复制算法到框架。
if(CCFW_LINK_EXISTING)
    find_package(OpenSSL REQUIRED)
    find_package(Threads REQUIRED)
    if(CCFW_BUILD_GUI AND WIN32 AND MINGW)
        # Only expose the C library headers from MSYS. Exposing its whole
        # include tree mixes MinGW 16 CRT headers into Qt's MinGW 13 compiler.
        set(CCFW_C_HEADERS "${CMAKE_BINARY_DIR}/native-dependency-headers")
        file(MAKE_DIRECTORY "${CCFW_C_HEADERS}")
        file(COPY "${OPENSSL_INCLUDE_DIR}/openssl" DESTINATION "${CCFW_C_HEADERS}")
        foreach(header sqlite3.h sqlite3ext.h)
            configure_file("${SQLite3_INCLUDE_DIR}/${header}" "${CCFW_C_HEADERS}/${header}" COPYONLY)
        endforeach()
        # QtGui also exposes Vulkan headers. Keep their include root isolated
        # from the other compiler's CRT just like OpenSSL and SQLite above.
        find_package(Vulkan QUIET)
        if(Vulkan_INCLUDE_DIR)
            foreach(directory vulkan vk_video)
                if(EXISTS "${Vulkan_INCLUDE_DIR}/${directory}")
                    file(COPY "${Vulkan_INCLUDE_DIR}/${directory}" DESTINATION "${CCFW_C_HEADERS}")
                endif()
            endforeach()
            set(Vulkan_INCLUDE_DIR "${CCFW_C_HEADERS}")
        endif()
        set_property(TARGET OpenSSL::Crypto PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${CCFW_C_HEADERS}")
        set_property(TARGET OpenSSL::SSL PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${CCFW_C_HEADERS}")
        get_target_property(CCFW_SQLITE_TARGET SQLite3::SQLite3 ALIASED_TARGET)
        if(NOT CCFW_SQLITE_TARGET)
            set(CCFW_SQLITE_TARGET SQLite3::SQLite3)
        endif()
        set_property(TARGET ${CCFW_SQLITE_TARGET} PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${CCFW_C_HEADERS}")
    endif()
    add_library(ccfw_existing STATIC
        "${CIPHERCHAT_ROOT}/common/auth/login_proof.cpp"
        "${CIPHERCHAT_ROOT}/common/base64.cpp"
        "${CIPHERCHAT_ROOT}/common/crypto/crypto.cpp"
        "${CIPHERCHAT_ROOT}/common/protocol/frame.cpp"
        "${CIPHERCHAT_ROOT}/common/protocol/messages.cpp"
        )
    target_link_libraries(ccfw_existing PUBLIC ccfw_contracts OpenSSL::Crypto Threads::Threads)
endif()

function(ccfw_module name)
    add_library(${name} STATIC ${ARGN})
    target_link_libraries(${name} PUBLIC ccfw_contracts)
endfunction()
ccfw_module(ccfw_storage src/local_store.cpp src/secret_protector.cpp)
target_link_libraries(ccfw_storage PUBLIC SQLite3::SQLite3 ccfw_existing)
if(WIN32)
    target_link_libraries(ccfw_storage PRIVATE crypt32)
endif()
ccfw_module(ccfw_transport src/transport.cpp)
target_sources(ccfw_transport PRIVATE
    "${CIPHERCHAT_ROOT}/client/core/net_client.cpp"
    "${CIPHERCHAT_ROOT}/common/net/tls_context.cpp")
target_link_libraries(ccfw_transport PUBLIC ccfw_existing OpenSSL::SSL)
ccfw_module(ccfw_updates update/update.cpp)
target_link_libraries(ccfw_updates PUBLIC ccfw_transport)
if(WIN32)
    add_executable(cipherchat-updater update/main.cpp)
    target_link_libraries(cipherchat-updater PRIVATE ccfw_updates)
    target_link_libraries(cipherchat-updater PRIVATE shell32 ole32)
    add_executable(cipherchat-launcher WIN32 update/launcher.cpp)
    target_link_libraries(cipherchat-launcher PRIVATE shell32)
    if(MINGW)
        target_link_options(cipherchat-updater PRIVATE -municode)
        target_link_options(cipherchat-launcher PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
endif()
if(WIN32)
    target_link_libraries(ccfw_transport PRIVATE ws2_32)
endif()

if(CCFW_BUILD_GUI)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
    # Keep the selected compiler runtime ahead of Qt and all Qt DLLs ahead of
    # MSYS Qt. The desktop preset uses Qt's matching MinGW 13.1 toolchain.
    if(WIN32 AND MINGW)
        get_filename_component(CCFW_COMPILER_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
        file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/compiler-runtime")
        foreach(runtime libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
            configure_file("${CCFW_COMPILER_DIRECTORY}/${runtime}" "${CMAKE_BINARY_DIR}/compiler-runtime/${runtime}" COPYONLY)
        endforeach()
    endif()
    # Codex-only desktop client: Widgets alone, no Qt Quick / Quick 3D runtime.
    add_library(ccfw_login_preferences STATIC gui/login_preferences.cpp)
    set_target_properties(ccfw_login_preferences PROPERTIES AUTOMOC ON)
    target_link_libraries(ccfw_login_preferences PUBLIC ccfw_client_runtime Qt6::Core)
    target_include_directories(ccfw_login_preferences PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/gui")
    add_library(ccfw_gui STATIC gui/main_window.cpp gui/theme.cpp gui/update_controller.cpp gui/feature_ui.cpp gui/local_file_actions.cpp gui/pencil_skin.cpp "${CIPHERCHAT_ROOT}/third_party/qrcodegen/qrcodegen.cpp")
    set_target_properties(ccfw_gui PROPERTIES AUTOMOC ON)
    if(MINGW)
        set_source_files_properties(gui/main_window.cpp PROPERTIES COMPILE_OPTIONS "-Wa,-mbig-obj")
    endif()
    target_link_libraries(ccfw_gui PUBLIC ccfw_client_runtime ccfw_login_preferences ccfw_updates Qt6::Widgets)
    if(WIN32)
        target_link_libraries(ccfw_gui PRIVATE shell32 ole32)
    endif()
    target_include_directories(ccfw_gui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/gui")
    add_executable(ccfw-desktop gui/main.cpp)
    target_link_libraries(ccfw-desktop PRIVATE ccfw_gui)
    set_target_properties(ccfw-desktop PROPERTIES WIN32_EXECUTABLE TRUE OUTPUT_NAME cipherchat)
endif()
